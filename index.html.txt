                `;
                return;
            }

            items.forEach(res => {
                let badgeClass = 'badge-paper';
                if (res.type === 'Textbook') badgeClass = 'badge-book';
                if (res.type === 'Study Note') badgeClass = 'badge-note';

                const card = document.createElement('div');
                card.className = 'resource-card';
                card.innerHTML = `
                    <div>
                        <div class="card-top">
                            <span class="badge ${badgeClass}">${res.type}</span>
                            <span class="resource-year">${res.year}</span>
                        </div>
                        <h3 class="resource-title">${res.title}</h3>
                        <div class="resource-info">
                            <span><i class="fa-solid fa-graduation-cap"></i> ${res.grade}</span>
                            <span><i class="fa-solid fa-book"></i> ${res.subject}</span>
                        </div>
                    </div>
                    <div class="card-actions">
                        <button class="btn btn-outline btn-sm" onclick="openPreviewModal(${res.id})">
                            <i class="fa-solid fa-eye"></i> Preview
                        </button>
                        <a href="${res.fileUrl}" target="_blank" class="btn btn-gold btn-sm" download>
                            <i class="fa-solid fa-download"></i> Download
                        </a>
                    </div>
                `;
                container.appendChild(card);
            });
        }

        /* FILTER SYSTEM */
        function setupFilterListeners() {
            const keywordInput = document.getElementById('filterKeyword');
            const typeSelect = document.getElementById('filterType');
            const gradeSelect = document.getElementById('filterGrade');
            const yearSelect = document.getElementById('filterYear');

            const applyFilters = () => {
                const keyword = keywordInput.value.toLowerCase();
                const type = typeSelect.value;
                const grade = gradeSelect.value;
                const year = yearSelect.value;

                const filtered = resourcesData.filter(res => {
                    const matchesKeyword = res.title.toLowerCase().includes(keyword) || res.subject.toLowerCase().includes(keyword);
                    const matchesType = type === 'All' || res.type === type;
                    const matchesGrade = grade === 'All' || res.grade === grade;
                    const matchesYear = year === 'All' || res.year === year;

                    return matchesKeyword && matchesType && matchesGrade && matchesYear;
                });

                renderResources(filtered);
            };

            keywordInput.addEventListener('input', applyFilters);
            typeSelect.addEventListener('change', applyFilters);
            gradeSelect.addEventListener('change', applyFilters);
            yearSelect.addEventListener('change', applyFilters);
        }

        function resetFilters() {
            document.getElementById('filterKeyword').value = '';
            document.getElementById('filterType').value = 'All';
            document.getElementById('filterGrade').value = 'All';
            document.getElementById('filterYear').value = 'All';
            renderResources(resourcesData);
        }

        function setFilter(type) {
            document.getElementById('filterType').value = type;
            const event = new Event('change');
            document.getElementById('filterType').dispatchEvent(event);
        }

        function quickSearch(tag) {
            document.getElementById('filterKeyword').value = tag;
            const event = new Event('input');
            document.getElementById('filterKeyword').dispatchEvent(event);
            document.getElementById('resources').scrollIntoView({ behavior: 'smooth' });
        }

        function triggerHeroSearch() {
            const val = document.getElementById('heroSearchInput').value;
            quickSearch(val);
        }

        function switchResourceCategory(type, element) {
            document.querySelectorAll('.resource-tabs .res-tab').forEach(tab => tab.classList.remove('active'));
            element.classList.add('active');
            setFilter(type);
        }

        /* MODAL HANDLERS */
        function openPreviewModal(id) {
            const item = resourcesData.find(r => r.id === id);
            if (!item) return;

            document.getElementById('modalTitle').textContent = item.title;
            document.getElementById('modalMeta').textContent = `${item.grade} • ${item.subject} • Medium: ${item.medium}`;
            document.getElementById('modalBadge').textContent = item.type;
            document.getElementById('modalDownloadBtn').href = item.fileUrl;

            document.getElementById('previewModal').classList.add('active');
        }

        function openUploadModal() {
            document.getElementById('uploadModal').classList.add('active');
        }

        function closeModal(modalId) {
            document.getElementById(modalId).classList.remove('active');
        }

        /* ACCORDION */
        function toggleAccordion(header) {
            const item = header.parentElement;
            item.classList.toggle('active');
        }

        /* TOAST NOTIFICATION */
        function showToast(message) {
            const toast = document.getElementById('toast');
            document.getElementById('toastMsg').textContent = message;
            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3500);
        }

        /* FORM SUBMISSIONS */
        function handleContactSubmit(e) {
            e.preventDefault();
            showToast("Thank you! Your message has been sent successfully.");
            e.target.reset();
        }

        function handleUploadSubmit(e) {
            e.preventDefault();
            closeModal('uploadModal');
            showToast("Resource submitted for teacher verification!");
            e.target.reset();
        }
